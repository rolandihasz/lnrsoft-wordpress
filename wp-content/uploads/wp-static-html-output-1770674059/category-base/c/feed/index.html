#include <algorithm>
#include <iostream>

constexpr unsigned exp = 31; // exponent
constexpr unsigned twoToExp = 1u << exp;

int solve_b_even(unsigned int n, int a, int b, int c)
{
    if (b == 0) {
        if (a == c) {
            return 1;
        }
        return 2;
    }
    if (a == 0 && c == 0) {
        return 1;
    }
    int a1_minus_a0 = (b - 1) * a + c;
    int numr = exp - __builtin_popcount((a1_minus_a0 & -a1_minus_a0) - 1);
    int denomr = __builtin_popcount((b & -b) - 1);
    unsigned int m = numr / denomr + (numr % denomr == 0 ? 1 : 2);
    return std::min(m, n);
}

int solve_b_odd(unsigned int n, int a, int b, int c)
{
    if (b == 1) {
        return c == 0 ? 1 : std::min(n, twoToExp / (c & -c));
    }
    if (a == 0 && c == 0) {
        return 1;
    }
    unsigned int m = 1;
    int b_minus_1 = b - 1;
    int a1_minus_a0 = b_minus_1 * a + c;
    int b_to_m = b;
    int mask = (twoToExp * (b_minus_1 & -b_minus_1) /
            (a1_minus_a0 & -a1_minus_a0)) -
           1;
    while (m < n && (b_to_m & mask) != 1) {
        b_to_m = b_to_m * b_to_m;
        m = m * 2;
    }
    return std::min(m, n);
}

int solve(unsigned int n, int a, int b, int c)
{
    if (b % 2 == 0) {
        return solve_b_even(n, a, b, c);
    }
    return solve_b_odd(n, a, b, c);
}

int main()
{
    // int n, a, b, c;
    int n = 32;
    int a = 16;
    int b = 16;
    int c = 16;
    // std::cin >> n >> a >> b >> c;
    std::cout << solve(n, a, b, c);
    return 0;
}

#include <fstream>
#include <limits.h>

using namespace std;

    cout << fibo[0] << " ";

    for (int i=1; i<INT_MAX; i++)
    {

        double n = fibo[i] + (fibo[ii]);
        fibo.push_back(n);

        }
        else
            return 0;
    }
